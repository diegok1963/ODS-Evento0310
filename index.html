// ============================================================ // GOOGLE APPS SCRIPT – Sistema de Reservas ODS // Almuerzo de Camaradería · 03/08/2026 // ============================================================ const SHEET_DATOS = 'ODS Evento 03-10-2026(Respuestas)'; const SHEET_DASHBOARD = 'Dashboard'; const TOTAL_CUPOS = 300; // ── Columnas de la hoja Reservas ──────────────────────────── const COL_TIMESTAMP = 1; // A - Fecha y hora const COL_PROMOCION = 2; // B - Número de Promoción const COL_NOMBRE = 3; // C - Nombre y Apellido const COL_FAMILIA = 4; // D - ¿Venís con Familia o Amigos? const COL_EMAIL = 5; // E - Correo Electrónico const COL_CELULAR = 6; // F - Número de Celular const COL_ACOMPANANTES= 7; // G - Acompañantes (detalle) const COL_CANT_ACOMP = 8; // H - Cantidad de acompañantes // ============================================================ // GET – contar reservas (incluye acompañantes) // ============================================================ function doGet(e) { const action = e.parameter.action; if (action === 'count') { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_DATOS); const totalFilas = Math.max(0, sheet.getLastRow() - 1); // Sumar titular + acompañantes let totalPersonas = 0; if (totalFilas > 0) { const cantAcomp = sheet.getRange(2, COL_CANT_ACOMP, totalFilas, 1).getValues().flat(); cantAcomp.forEach(c => { totalPersonas += 1 + (parseInt(c) || 0); }); } return jsonResponse({ count: totalPersonas, reservas: totalFilas, cuposRestantes: Math.max(0, TOTAL_CUPOS - totalPersonas) }); } return jsonResponse({ status: 'ok', message: 'ODS Reservas API activa' }); } // ============================================================ // POST – guardar reserva // ============================================================ function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.action === 'reservar') { return procesarReserva(payload); } return jsonResponse({ status: 'error', message: 'Acción no reconocida' }); } catch (err) { return jsonResponse({ status: 'error', message: err.toString() }); } } // ============================================================ // PROCESAR RESERVA // ============================================================ function procesarReserva(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = obtenerOCrearHojaReservas(ss); const acompanantes = data.acompanantes || []; const cantAcompanantes = acompanantes.length; const totalNuevos = 1 + cantAcompanantes; // titular + acompañantes // 1. Calcular cupos usados actuales const totalFilas = Math.max(0, sheet.getLastRow() - 1); let cuposUsados = 0; if (totalFilas > 0) { const cantAcomp = sheet.getRange(2, COL_CANT_ACOMP, totalFilas, 1).getValues().flat(); cantAcomp.forEach(c => { cuposUsados += 1 + (parseInt(c) || 0); }); } // 2. Verificar cupos disponibles if (cuposUsados + totalNuevos > TOTAL_CUPOS) { return jsonResponse({ status: 'sin_cupos', cuposRestantes: Math.max(0, TOTAL_CUPOS - cuposUsados) }); } // 3. Verificar duplicado por email if (totalFilas > 0) { const emails = sheet.getRange(2, COL_EMAIL, totalFilas, 1).getValues().flat(); const emailNorm = data.email.trim().toLowerCase(); if (emails.some(e => e.toString().trim().toLowerCase() === emailNorm)) { return jsonResponse({ status: 'duplicado' }); } } // 4. Formatear acompañantes como texto let acompTexto = ''; if (cantAcompanantes > 0) { acompTexto = acompanantes.map((a, i) => `${i+1}. ${a.nombre} ${a.apellido} (${a.relacion})` ).join(' | '); } // 5. Guardar la fila const timestamp = Utilities.formatDate( new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy HH:mm:ss' ); sheet.appendRow([ timestamp, 'Promoción ' + data.promocion, data.nombre.trim(), data.familia, data.email.trim().toLowerCase(), data.celular.trim(), acompTexto, cantAcompanantes ]); // 6. Actualizar dashboard actualizarDashboard(ss); return jsonResponse({ status: 'ok', message: '¡Reserva registrada con éxito!', totalPersonas: totalNuevos }); } // ============================================================ // CREAR / VERIFICAR HOJA DE RESERVAS // ============================================================ function obtenerOCrearHojaReservas(ss) { let sheet = ss.getSheetByName(SHEET_DATOS); if (!sheet) { sheet = ss.insertSheet(SHEET_DATOS); } // Agregar columnas nuevas si faltan (para hojas existentes) if (sheet.getLastRow() > 0) { const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (headers.length < 8) { // Agregar encabezados faltantes if (headers.length < 7) sheet.getRange(1, 7).setValue('Acompañantes'); if (headers.length < 8) sheet.getRange(1, 8).setValue('Cantidad Acompañantes'); // Formato const newHeaders = sheet.getRange(1, 7, 1, 2); newHeaders.setBackground('#002366'); newHeaders.setFontColor('#FFFFFF'); newHeaders.setFontWeight('bold'); newHeaders.setFontSize(11); sheet.setColumnWidth(7, 350); sheet.setColumnWidth(8, 100); } return sheet; } // Hoja nueva: crear encabezados completos const headers = [ 'Fecha y Hora', 'Número de Promoción', 'Nombre y Apellido', '¿Venís con Familia o Amigos (No Liciestas)?', 'Tu Correo Electrónico', 'Tu Número de Celular', 'Acompañantes', 'Cantidad Acompañantes' ]; sheet.appendRow(headers); const headerRange = sheet.getRange(1, 1, 1, headers.length); headerRange.setBackground('#002366'); headerRange.setFontColor('#FFFFFF'); headerRange.setFontWeight('bold'); headerRange.setFontSize(11); sheet.setColumnWidth(1, 160); sheet.setColumnWidth(2, 160); sheet.setColumnWidth(3, 200); sheet.setColumnWidth(4, 260); sheet.setColumnWidth(5, 220); sheet.setColumnWidth(6, 160); sheet.setColumnWidth(7, 350); sheet.setColumnWidth(8, 100); sheet.setFrozenRows(1); return sheet; } // ============================================================ // ACTUALIZAR DASHBOARD // ============================================================ function actualizarDashboard(ss) { let dash = ss.getSheetByName(SHEET_DASHBOARD); if (!dash) dash = ss.insertSheet(SHEET_DASHBOARD); dash.clearContents(); const sheet = ss.getSheetByName(SHEET_DATOS); const totalFilas = Math.max(0, sheet.getLastRow() - 1); let totalPersonas = 0; let totalAcompanantes = 0; if (totalFilas > 0) { const cantAcomp = sheet.getRange(2, COL_CANT_ACOMP, totalFilas, 1).getValues().flat(); cantAcomp.forEach(c => { const n = parseInt(c) || 0; totalAcompanantes += n; totalPersonas += 1 + n; }); } const cuposRestantes = Math.max(0, TOTAL_CUPOS - totalPersonas); const pct = totalPersonas > 0 ? ((totalPersonas / TOTAL_CUPOS) * 100).toFixed(1) : '0.0'; // Título dash.getRange('A1').setValue('DASHBOARD – RESERVAS ALMUERZO ODS 2026'); dash.getRange('A1').setFontSize(14).setFontWeight('bold').setFontColor('#002366'); dash.getRange('A1:D1').merge(); dash.getRange('A2').setValue('Actualizado: ' + Utilities.formatDate( new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM/yyyy HH:mm' )); dash.getRange('A2').setFontColor('#888888').setFontSize(10); dash.getRange('A2:D2').merge(); // Métricas dash.getRange('A4').setValue('📊 RESUMEN GENERAL'); dash.getRange('A4').setFontWeight('bold').setFontColor('#002366').setFontSize(12); dash.getRange('A4:D4').merge(); const metricas = [ ['Total Cupos', TOTAL_CUPOS, 'Cupos Ocupados', totalPersonas], ['Cupos Restantes', cuposRestantes, 'Ocupación (%)', pct + '%'], ['Reservas (titulares)', totalFilas,'Total Acompañantes', totalAcompanantes], ]; metricas.forEach((row, i) => { dash.getRange(5 + i, 1, 1, 4).setValues([row]); dash.getRange(5 + i, 1, 1, 4).setBackground(i % 2 === 0 ? '#F0F4FF' : '#FFFFFF'); }); dash.getRange('B5:B7').setFontWeight('bold').setFontSize(13).setFontColor('#002366'); dash.getRange('D5:D7').setFontWeight('bold').setFontSize(13).setFontColor('#16A34A'); // Reservas por Promoción if (totalFilas > 0) { const datos = sheet.getRange(2, COL_PROMOCION, totalFilas, 1).getValues().flat(); const conteo = {}; datos.forEach(p => { conteo[p] = (conteo[p] || 0) + 1; }); const sorted = Object.entries(conteo).sort((a, b) => { return parseInt(a[0].replace('Promoción ','')) - parseInt(b[0].replace('Promoción ','')); }); dash.getRange('A9').setValue('📋 RESERVAS POR PROMOCIÓN'); dash.getRange('A9').setFontWeight('bold').setFontColor('#002366').setFontSize(12); dash.getRange('A9:B9').merge(); dash.getRange('A10').setValue('Promoción'); dash.getRange('B10').setValue('Reservas'); dash.getRange('A10:B10').setBackground('#002366').setFontColor('#FFFFFF').setFontWeight('bold'); sorted.forEach(([prom, cant], i) => { dash.getRange(11 + i, 1).setValue(prom); dash.getRange(11 + i, 2).setValue(cant); if (i % 2 === 0) dash.getRange(11 + i, 1, 1, 2).setBackground('#F8FAFF'); }); // Acompañantes const familiaData = sheet.getRange(2, COL_FAMILIA, totalFilas, 1).getValues().flat(); const conFamilia = familiaData.filter(v => v === 'Sí').length; const sinFamilia = familiaData.filter(v => v === 'No').length; const sr = 12 + sorted.length; dash.getRange(sr, 1).setValue('👨‍👩‍👧 ACOMPAÑANTES'); dash.getRange(sr, 1).setFontWeight('bold').setFontColor('#002366').setFontSize(12); dash.getRange(sr, 1, 1, 2).merge(); dash.getRange(sr+1, 1).setValue('Vienen acompañados'); dash.getRange(sr+1, 2).setValue(conFamilia); dash.getRange(sr+2, 1).setValue('Vienen solos'); dash.getRange(sr+2, 2).setValue(sinFamilia); dash.getRange(sr+3, 1).setValue('Total acompañantes'); dash.getRange(sr+3, 2).setValue(totalAcompanantes); dash.getRange(sr+1, 1, 3, 2).setBackground('#F0FDF4'); } dash.setColumnWidth(1, 220); dash.setColumnWidth(2, 130); dash.setColumnWidth(3, 220); dash.setColumnWidth(4, 130); } // ============================================================ // HELPER – Respuesta JSON // ============================================================ function jsonResponse(obj) { const output = ContentService.createTextOutput(JSON.stringify(obj)); output.setMimeType(ContentService.MimeType.JSON); return output; } // ============================================================ // FUNCIÓN MANUAL – Inicializar hojas // ============================================================ function inicializarHojas() { const ss = SpreadsheetApp.getActiveSpreadsheet(); obtenerOCrearHojaReservas(ss); actualizarDashboard(ss); SpreadsheetApp.getUi().alert('✅ Hojas inicializadas correctamente.'); }
